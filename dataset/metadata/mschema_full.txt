# Table: trd_order, 订单主表：一笔用户下单一行，金额字段口径见业务知识
[
(id:BIGINT, Primary Key, 订单ID),
(order_no:VARCHAR, 订单编号),
(member_id:BIGINT, 下单会员ID),
(shop_id:BIGINT, 店铺ID),
(source_id:INT, 订单来源ID),
(order_status:VARCHAR, 订单状态：CREATED待支付/PAID已支付/SHIPPED已发货/COMPLETED已完成/CANCELLED已取消/CLOSED已关闭),
(total_amount:DECIMAL, 商品总额（原价合计，不含运费，未扣优惠）),
(discount_amount:DECIMAL, 优惠总额（券+活动+会员折扣）),
(freight_amount:DECIMAL, 运费),
(pay_amount:DECIMAL, 实付金额=商品总额-优惠总额+运费),
(currency_code:VARCHAR, 币种),
(created_at:DATETIME, 下单时间),
(paid_at:DATETIME, 支付时间，未支付为空),
(completed_at:DATETIME, 确认收货时间),
(is_presale:TINYINT, 是否预售订单 1是0否),
(pay_type:VARCHAR, 支付方式：ONLINE/COD/INSTALLMENT),
(device_type:VARCHAR, 下单设备：IOS/ANDROID/PC/H5),
(buyer_remark:VARCHAR, 买家备注),
(seller_remark:VARCHAR, 卖家备注),
(invoice_flag:TINYINT, 是否需要发票),
(points_used:INT, 使用积分数),
(points_amount:DECIMAL, 积分抵扣金额),
(tax_amount:DECIMAL, 税费),
(order_type:VARCHAR, 订单类型：NORMAL/PRESALE/GROUP_BUY/EXCHANGE),
(is_first_order:TINYINT, 是否首单),
(risk_level:VARCHAR, 风控等级：LOW/MEDIUM/HIGH),
(close_reason:VARCHAR, 关闭原因：STOCKOUT/TIMEOUT/FRAUD/USER),
(delivery_type:VARCHAR, 配送方式：EXPRESS/SAME_DAY/PICKUP),
(expected_delivery_date:DATE, 预计送达日期),
(created_by:VARCHAR, 创建人),
(created_time:DATETIME, 记录创建时间),
(updated_time:DATETIME, 记录更新时间),
(is_deleted:TINYINT, 逻辑删除标记 1已删除0正常)
]
